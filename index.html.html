Configuração Inicial
Siga os passos abaixo para conectar o sistema ao Supabase. Você só precisa fazer isso uma vez.
Passo 1 — Criar projeto no Supabase
  1. Acesse supabase.com e clique em Start your project
  2. Crie uma conta gratuita com seu e-mail (ou Google)
  3. Clique em New Project, escolha um nome (ex: parente-andrade-rh) e uma senha forte
  4. Aguarde o projeto ser criado (cerca de 1 minuto)
Passo 2 — Criar as tabelas no banco

No painel do Supabase, vá em SQL Editor → New query, cole o código abaixo e clique em Run:

-- Tabela de colaboradores create table if not exists colaboradores ( id uuid primary key default gen_random_uuid(), nome text not null, cpf text not null, funcao text not null, centro_custo text not null, numero_rpmqp text, tipo_doc text default 'RP', mqp_subtipo text, mqp_cargo_atual text, mqp_cargo_final text, mqp_salario_atual text, mqp_salario_final text, mqp_cc_atual text, mqp_cc_destino text, mqp_funcao_atual text, mqp_funcao_final text, criado_em timestamptz default now() ); -- Colunas MQP (para bancos existentes) alter table colaboradores add column if not exists mqp_subtipo text; alter table colaboradores add column if not exists mqp_cargo_atual text; alter table colaboradores add column if not exists mqp_cargo_final text; alter table colaboradores add column if not exists mqp_salario_atual text; alter table colaboradores add column if not exists mqp_salario_final text; alter table colaboradores add column if not exists mqp_cc_atual text; alter table colaboradores add column if not exists mqp_cc_destino text; alter table colaboradores add column if not exists mqp_funcao_atual text; alter table colaboradores add column if not exists mqp_funcao_final text; -- Tabela de documentos create table if not exists documentos ( id uuid primary key default gen_random_uuid(), colaborador_id uuid references colaboradores(id) on delete cascade, doc_id text not null, status text default 'none', na boolean default false, arquivo_url text, arquivo_nome text, atualizado_em timestamptz default now() ); -- Storage bucket para arquivos insert into storage.buckets (id, name, public) values ('documentos', 'documentos', true) on conflict do nothing; -- Permissões de leitura/escrita públicas (ajuste conforme necessário) create policy if not exists "allow_all_documentos" on storage.objects for all using (bucket_id = 'documentos'); alter table colaboradores enable row level security; alter table documentos enable row level security; create policy if not exists "allow_all_colab" on colaboradores for all using (true); create policy if not exists "allow_all_docs" on documentos for all using (true); -- Tabela de fluxo pipeline (CC 1238/1217) create table if not exists fluxo_pipeline ( id uuid primary key default gen_random_uuid(), colaborador_id uuid references colaboradores(id) on delete cascade, etapa text not null, status text default 'none', obs text, atualizado_em timestamptz default now(), unique(colaborador_id, etapa) ); alter table fluxo_pipeline add column if not exists obs text; alter table fluxo_pipeline enable row level security; create policy if not exists "allow_all_fluxo" on fluxo_pipeline for all using (true);
